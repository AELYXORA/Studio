cmake_minimum_required(VERSION 3.24)
project(AELYXORAStudio LANGUAGES CXX)
include(CTest)
add_library(aelyxora_model STATIC src/model.cpp src/transaction.cpp)
add_library(aelyxora_properties STATIC src/properties.cpp src/registry.cpp src/bindings.cpp)
target_include_directories(aelyxora_properties PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(aelyxora_properties PUBLIC aelyxora_model)
target_compile_features(aelyxora_properties PUBLIC cxx_std_23)
target_include_directories(aelyxora_model PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(aelyxora_model PUBLIC cxx_std_23)
if(MSVC)
  target_compile_options(aelyxora_model PRIVATE /W4 /permissive-)
else()
  target_compile_options(aelyxora_model PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(model_tests tests/model_tests.cpp)
  target_link_libraries(model_tests PRIVATE aelyxora_model)
  add_test(NAME model_contracts COMMAND model_tests)
  add_executable(transaction_tests tests/transaction_tests.cpp)
  target_link_libraries(transaction_tests PRIVATE aelyxora_model)
  add_test(NAME transaction_contracts COMMAND transaction_tests)
  add_executable(properties_tests tests/properties_tests.cpp)
  target_link_libraries(properties_tests PRIVATE aelyxora_properties)
  add_test(NAME typed_properties_contracts COMMAND properties_tests)
  add_executable(registry_tests tests/registry_tests.cpp)
  target_link_libraries(registry_tests PRIVATE aelyxora_properties)
  add_test(NAME component_registry_contracts COMMAND registry_tests)
  add_executable(bindings_tests tests/bindings_tests.cpp)
  target_link_libraries(bindings_tests PRIVATE aelyxora_properties)
  add_test(NAME binding_graph_contracts COMMAND bindings_tests)
endif()

# Experimental persistence is opt-in; core/Windows CI do not require Boost.
option(AELYXORA_WITH_BOOST_JSON "Build Boost.JSON persistence adapter (Boost 1.81+)" OFF)
if(AELYXORA_WITH_BOOST_JSON)
  find_package(Boost 1.81 REQUIRED)
  add_library(aelyxora_persistence STATIC src/persistence.cpp)
  target_include_directories(aelyxora_persistence PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include ${Boost_INCLUDE_DIRS})
  target_link_libraries(aelyxora_persistence PUBLIC aelyxora_model)
  target_compile_features(aelyxora_persistence PUBLIC cxx_std_23)
  if(BUILD_TESTING)
    add_executable(persistence_tests tests/persistence_tests.cpp)
    target_link_libraries(persistence_tests PRIVATE aelyxora_persistence)
    add_test(NAME persistence_contracts COMMAND persistence_tests)
  endif()
endif()
